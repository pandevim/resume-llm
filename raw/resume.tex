\section{Graduate Research}

\noindent
\begin{minipage}[t]{0.94\textwidth}\raggedright
    \textbf{Self-Aware Resume}: 83M hybrid model (attention and \tech{Mamba-3} in parallel) with a copy head that quotes from the document in context; pretrained on 100B tokens on 4$\times$\tech{H100}, then tuned with \tech{SFT} and \tech{GRPO}. Scores 0.80 on resume QA vs 0.15 for \tech{Falcon-H1-Tiny-90M} and \tech{SmolLM2-135M}; runs on a single CPU core in under 1 GB of memory (0.14 s per token) via \tech{ONNX Runtime}.

    \medskip
    \textbf{\href{https://github.com/pandevim/TriMem}{Tri-Mem}}: Modality-matched memory for multi-agent systems; semantic memory (sparse-attention KV chunks), episodic memory (visual compression), and declarative memory (vector RAG) with entropy-driven routing. Cuts token cost by 84\% on a custom NovaCorp AuditBench and reaches 81.4\% on \tech{LongMemEval}. \textit{(Presented at the \href{https://www.stevens.edu/icns-center-for-innovative-computing-and-networked-systems/symposium\#spring-2026}{ICNS Symposium}, Spring 2026)}

    \medskip
    \textbf{NeuralSINDy}: Extending \tech{SINDy} with neural function approximators to discover governing equations from noisy, high-dimensional dynamical systems data. \textit{(In preparation)}
